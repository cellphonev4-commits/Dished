        // A basic function to convert US to Metric
        function convertToMetric(ingredientText) {
            let s = ingredientText;

            // First, change fractions to decimals to make math easy
            s = s.replace("1/2", "0.5");
            s = s.replace("1/4", "0.25");
            s = s.replace("3/4", "0.75");

            // Replace cups with ml (1 cup = 240 ml)
            s = s.replace(/([0-9.]+)\s*cup[s]?/gi, function (match, number) { return (number * 240) + " ml"; });
            // Replace tbsp with ml (1 tbsp = 15 ml)
            s = s.replace(/([0-9.]+)\s*tbsp/gi, function (match, number) { return (number * 15) + " ml"; });
            // Replace tsp with ml (1 tsp = 5 ml)
            s = s.replace(/([0-9.]+)\s*tsp/gi, function (match, number) { return (number * 5) + " ml"; });
            // Replace lb with g (1 lb = 450 g)
            s = s.replace(/([0-9.]+)\s*lb/gi, function (match, number) { return (number * 450) + " g"; });
            // Replace oz with g (1 oz = 28 g)
            s = s.replace(/([0-9.]+)\s*oz/gi, function (match, number) { return (number * 28) + " g"; });

            return s;
        }

                    const li = document.createElement('li');
